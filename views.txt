  label: string
  options: readonly string[]
  value?: string
  any?: string
  required?: boolean
}> = ({ name, label, options, value, any, required }) => (
  <label>
    <span>{label}</span>
    <select name={name} required={required}>
      {any !== undefined && <option value="">{any}</option>}
      {options.map((o) => (
        <option value={o} selected={o === value}>
          {LABELS[o]}
        </option>
      ))}
    </select>
  </label>
)

const SpecializationTags: FC<{ ids: string[]; limit?: number }> = ({ ids, limit }) => {
  const shown = limit ? ids.slice(0, limit) : ids
  const rest = limit ? ids.length - shown.length : 0
  return (
    <ul class="tags">
      {shown.map((id) => (
        <li>{specLabel(id)}</li>
      ))}
      {rest > 0 && <li>+{rest}</li>}
    </ul>
  )
}

const TeamList: FC<{ teams: Team[]; filtered?: boolean; showModeration?: boolean }> = ({ teams, filtered, showModeration }) =>
  teams.length === 0 ? (
    <div class="empty">
      {filtered ? (
        <>
          <p>لا توجد فرق مطابقة لخيارات التصفية.</p>
          <a class="btn quiet" href="/browse">مسح التصفية</a>
        </>
      ) : (
        <p>لا توجد فرق هنا بعد.</p>
      )}
    </div>
  ) : (
    <ul class="rows">
      {teams.map((t) => (
        <li class={`row status-${t.status}`}>
          <div class="need">
            <span class="need-type" title={LABELS[t.type]}>
              <i class={TYPE_ICONS[t.type] || 'fa-solid fa-circle'} aria-hidden="true"></i>
            </span>
            <span class="need-status">{LABELS[t.status]}</span>
          </div>
          <div class="main">
            <h2>
              <a href={`/t/${t.id}`}>{t.title}</a>
            </h2>
            {showModeration && t.approval_status !== 'approved' && (
              <div class={`note ${t.approval_status === 'rejected' ? '' : 'warn'}`}>
                حالة النشر: {LABELS[t.approval_status] || t.approval_status}
              </div>
            )}
            {showModeration && t.pending_changes && <div class="note warn">تعديلات بانتظار موافقة المشرف</div>}
            {t.description && <p class="blurb">{excerpt(t.description)}</p>}
            <div class="seek">
              <span class="seek-label">نبحث عن</span>
              {t.looking_for.length ? <SpecializationTags ids={t.looking_for} limit={4} /> : <span class="blurb">متخصصين</span>}
            </div>
            <div class="meta">
              <span>
                <i class="fa-solid fa-location-dot" aria-hidden="true"></i>
                {whereText(t)}
              </span>
              {t.creator_name && (
                <a class="creator" href={`/u/${t.creator_id}`}>
                  بواسطة {t.creator_name}
                </a>
              )}
            </div>
          </div>
        </li>
      ))}
    </ul>
  )

export const TeamPage: FC<{
  team: Team
  user: User | null
  csrfToken?: string | null
  canView: boolean
  myRequest: JoinRequest | null
  requests: JoinRequest[]
  members: JoinRequest[]
}> = ({ team, user, csrfToken, canView, myRequest, requests, members }) => {
  const display = teamWithPendingApplied(team)
  const isOwner = !!user && user.id === team.creator_id
  const isAdmin = isAdminUser(user)
  const pending = requests.filter((r) => r.status === 'pending')
  const isApproved = team.approval_status === 'approved'

  return (
    <Layout title={team.title} user={user} csrfToken={csrfToken}>
      <a class="back" href="/browse">
        تصفح الفرق
      </a>
      <div class="head">
        <div>
          <h1>{team.title}</h1>

          {team.creator_name && (
            <a class="byline" href={`/u/${team.creator_id}`}>
              فريق {team.creator_name}
            </a>
          )}
          {isOwner && <div class="note ok">أنت منشئ هذا الفريق</div>}
          {team.approval_status === 'pending' && (isOwner || isAdmin) && (
            <div class="note warn">هذا الفريق بانتظار موافقة المشرف ولن يظهر للآخرين حتى تتم الموافقة.</div>
          )}
          {team.approval_status === 'rejected' && (isOwner || isAdmin) && (
            <div class="note">تم رفض نشر هذا الفريق. يمكنك تعديله وإعادة إرساله للمراجعة.</div>
          )}
          {isApproved && team.pending_changes && (isOwner || isAdmin) && (
            <div class="note warn">هناك تعديلات بانتظار موافقة المشرف. الإصدار الحالي هو الظاهر للآخرين.</div>
          )}
          {!isOwner && isApproved && myRequest?.status === 'accepted' && <div class="note ok">أنت عضو في هذا الفريق</div>}
          {!isOwner && isApproved && myRequest?.status === 'pending' && <div class="note warn">طلب انضمامك قيد الانتظار</div>}
          {!isOwner && isApproved && myRequest?.status === 'rejected' && <div class="note">لم يُقبل طلبك السابق — يمكنك إعادة الطلب</div>}
        </div>
        {isOwner ? (
          <div class="actions">
            <a class="btn quiet" href={`/t/${team.id}/edit`}>
              تعديل الفريق
            </a>
            <form
              method="post"
              action={`/t/${team.id}/delete`}
              onsubmit="return confirm('هل تريد حذف هذا الفريق؟ لا يمكن التراجع عن ذلك.')"
            >
              {csrfToken ? <input type="hidden" name={CSRF_FIELD} value={csrfToken} /> : null}
              <button class="btn danger" type="submit">
                حذف الفريق
              </button>
            </form>
          </div>
        ) : !isOwner && myRequest?.status === 'accepted' ? (
          <div class="actions">
            <form
              method="post"
              action={`/t/${team.id}/leave`}
              onsubmit="return confirm('هل تريد مغادرة هذا الفريق؟ ستفقد الوصول حتى يتم قبولك مرة أخرى.')"
            >
              {csrfToken ? <input type="hidden" name={CSRF_FIELD} value={csrfToken} /> : null}
              <button class="btn danger" type="submit">
                مغادرة الفريق
              </button>
            </form>
          </div>
        ) : null}
      </div>

      {!canView ? (
        <div class="detail">
          <section>
            {team.description ? <p class="prose">{team.description}</p> : <p class="locked">لا يوجد وصف بعد.</p>}
            <div class="panel" style="margin-top:1.5rem">
              <h3>معلومات التواصل للأعضاء فقط</h3>
              <p class="locked" style="margin-bottom:1rem">
                يمكنك الاطلاع على معلومات التواصل بعد أن يقبل منشئ الفريق طلب انضمامك.
              </p>
              {!user ? (
                <a class="btn" href={`/login?next=/t/${team.id}`}>
                  سجّل الدخول لطلب الانضمام
                </a>
              ) : myRequest?.status === 'pending' ? (
                <p class="locked" style="margin:0">
                  طلبك بانتظار رد منشئ الفريق.
                </p>
              ) : isApproved ? (
                <form method="post" action={`/t/${team.id}/join`}>

          </section>
          <aside>
            <div class="panel">
              <dl>

const Avatar: FC<{ name: string }> = ({ name }) => {
  const first = [...name.trim()][0] ?? '؟'
  const idx = (first.codePointAt(0) ?? 0) % 3
  return (
    <span class={`avatar a${idx}`} aria-hidden="true">
      {first}
    </span>
  )
}

const formatPendingValue = (key: string, value: unknown): string => {
  if (value === null || value === undefined) return '—'
  if (key === 'online_allowed') return value ? 'نعم (متاح عن بُعد)' : 'لا (حضوري فقط)'
  if (key === 'looking_for' && Array.isArray(value)) {
    return value.length ? value.map((id) => specLabel(String(id))).join('، ') : '—'
  }
  if ((key === 'status' || key === 'type') && typeof value === 'string') return LABELS[value] || value
  return String(value)
}

const PendingChangesDetails: FC<{ team: Team }> = ({ team }) => {
  const changes = team.pending_changes
  if (!changes) return null
  const entries = Object.entries(changes).filter(([, v]) => v !== undefined)
  if (entries.length === 0) return null
  return (
    <div class="pending-changes">
      <h4>التعديلات المقترحة (القيم الجديدة)</h4>
      <dl>
        {entries.map(([key, value]) => (
          <>
            <dt>{pendingFieldLabel[key] || key}</dt>
            <dd>{formatPendingValue(key, value)}</dd>
          </>
        ))}
      </dl>
    </div>
  )
}

          <h2>تبحث عن أشخاص لفريقك؟</h2>
          <a href="/people">عرض كل الأشخاص</a>
        </div>
        <p class="section-sub">أشخاص متاحون للتواصل والتعاون. افتح ملفاتهم للتعرّف على مهاراتهم.</p>
        {people.length ? (
          <div class="people-grid">
            {people.map((p) => (
              <PersonCard u={p} />
            ))}
          </div>
        ) : (
          <div class="empty">
            <p>لا يوجد أشخاص متاحون حاليًا.</p>
            <a class="btn quiet" href="/people">
              تصفح صفحة الأشخاص
            </a>
          </div>
        )}
      </section>

      <section class="section">
        <div class="section-head">
          <h2>مسابقات وهاكاثونات قادمة</h2>
          <a href="/hackathons">عرض الكل</a>
        </div>
        <p class="section-sub">سجّل في مسابقة، ثم ابحث عن فريقك قبل موعد البداية.</p>
        <div class="event-list">
          {upcoming.map((h) => {
            const d = splitDate(h.start)
            return (
              <article class="event">
                <div class="event-date">
                  <b>{d.day}</b>
                  <span>{d.month}</span>
                </div>
                <div class="event-info">
                  <h3>{h.title}</h3>
                  <p>
                    {h.organizer}
                    {h.location ? `، ${h.location}` : ''}
                  </p>
                  {h.end !== h.start ? <p>حتى {h.end}</p> : null}
                </div>
                <a class="btn small quiet" href={h.url} target="_blank" rel="noopener noreferrer">
                  التفاصيل
                </a>
              </article>
            )
          })}
        </div>
      </section>
    </Layout>
  )
}
